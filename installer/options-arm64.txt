# OverLay-Trans 실행 옵션. 한 줄에 옵션 하나씩 "--이름 값" 형식으로 적습니다.
# #으로 시작하는 줄은 무시합니다. 고친 뒤 앱을 다시 실행하면 적용됩니다.
# 쓸 수 있는 옵션은 https://github.com/higashiaka/Overlay-Trans 의 README에 있습니다.

# 방송 언어. 지정하면 자동 감지(auto)보다 빠릅니다.
--language ja

# 음성 인식 모델. 스냅드래곤 X에서는 CPU로 인식하므로 작은 모델(small)을 씁니다.
# 더 빠르게 하려면 ggml-base.bin으로 바꾸세요. 대신 낱말을 더 자주 틀립니다.
# models 폴더에 없는 모델은 다음에 실행할 때 내려받습니다.
--stt-model models\ggml-small.bin

# 음성 인식이 한 번에 계산하는 길이. 기본값(768)의 절반으로 줄여 인식 시간을 절반으로 줄입니다.
# 짧은 감탄사를 놓치는 일이 늘면 이 줄을 지우세요.
--stt-audio-ctx 384
